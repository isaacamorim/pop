CREATE OR REPLACE TRIGGER TRG_I_POP_TEMPLATE_BI
BEFORE INSERT ON I_POP_TEMPLATE
FOR EACH ROW
BEGIN
  IF :NEW.IPT_ID IS NULL THEN
    SELECT SEQ_I_POP_TEMPLATE.NEXTVAL INTO :NEW.IPT_ID FROM DUAL;
  END IF;
END;
